01

{% trans "Create the Django project" %}

{% trans "Start with a fresh virtual environment and install Django. Then create the project and your app." %}

# Create virtual environment
python -m venv venv
venv\Scripts\activate        # Windows
# source venv/bin/activate  # macOS/Linux

pip install django

# Create project and app
django-admin startproject myproject .
python manage.py startapp myapp
02

{% trans "Configure internationalization in settings.py" %}

{% trans "Enable i18n, define supported languages, set the locale path, and register your app." %}

# settings.py

INSTALLED_APPS = [
    ...
    'myapp',
]

MIDDLEWARE = [
    'django.middleware.security.SecurityMiddleware',
    'django.contrib.sessions.middleware.SessionMiddleware',
    'django.middleware.locale.LocaleMiddleware',  # must be after SessionMiddleware
    'django.middleware.common.CommonMiddleware',
    ...
]

LANGUAGE_CODE = 'en'
USE_I18N = True

LANGUAGES = [
    ('en', 'English'),
    ('lv', 'Latvian'),
    ('fr', 'French'),
]

LOCALE_PATHS = [BASE_DIR / 'myapp/locale']

# settings.py → TEMPLATES → OPTIONS → context_processors

'django.template.context_processors.i18n',
04

{% trans "Wire up the language-switch URL" %}

{% trans "Django ships a built-in set_language view. Include it in your root urls.py." %}

# urls.py (project root)
from django.urls import path, include

urlpatterns = [
    path('i18n/', include('django.conf.urls.i18n')),
    path('', include('myapp.urls')),
]
05

{% trans "Mark strings for translation" %}

{% trans "In templates use the trans tag. In Python views use gettext. Only marked strings get extracted." %}

{# template.html #}
{% verbatim %}{% load i18n %}{% endverbatim %}
<h1>{% verbatim %}{% trans "Welcome to our site" %}{% endverbatim %}</h1>

# views.py
from django.utils.translation import gettext as _
message = _("Hello, world!")
06

{% trans "Build the dynamic language switcher" %}

{% trans "Loop over LANGUAGES from settings — when you add a language there, the dropdown updates automatically." %}

{# base.html navbar #}
{% verbatim %}{% load i18n %}{% endverbatim %}

<form method="post" action="{% verbatim %}{% url 'set_language' %}{% endverbatim %}">
  {% verbatim %}{% csrf_token %}{% endverbatim %}
  <input name="next" type="hidden" value="{% verbatim %}{{ request.get_full_path }}{% endverbatim %}">

  {% verbatim %}{% get_language_info_list for LANGUAGES as languages %}{% endverbatim %}
  <select name="language" onchange="this.form.submit()">
    {% verbatim %}{% for language in languages %}{% endverbatim %}
    <option value="{% verbatim %}{{ language.code }}{% endverbatim %}">
      {% verbatim %}{{ language.name_local }}{% endverbatim %}
    </option>
    {% verbatim %}{% endfor %}{% endverbatim %}
  </select>
</form>
07

{% trans "Extract and compile translation files" %}

{% trans "Run makemessages for each language, fill in translations in the .po files, then compile to .mo binaries." %}

# Extract strings for each language
python manage.py makemessages -l lv
python manage.py makemessages -l fr

# Edit: myapp/locale/lv/LC_MESSAGES/django.po
# Edit: myapp/locale/fr/LC_MESSAGES/django.po

# Compile .po → .mo (required for translations to work)
python manage.py compilemessages
{% trans "Note" %}: {% trans "On Windows you need GNU gettext tools installed. Download from gnuwin32 or use WSL." %}
08

{% trans "Add a new language in the future" %}

{% trans "Only three things are needed — no template changes required thanks to the dynamic dropdown." %}

# 1. Add to settings.py
LANGUAGES = [
    ...
    ('de', 'German'),   # new language
]

# 2. Extract & translate
python manage.py makemessages -l de
# edit locale/de/LC_MESSAGES/django.po
python manage.py compilemessages

# 3. Done — German appears in the navbar automatically
09

{% trans "Understand the request flow" %}

{% trans "When a user picks a language, this is what happens behind the scenes." %}

  1. {% trans "User selects a language → form POSTs to /i18n/setlang/" %}
  2. {% trans "set_language saves the choice in session/cookie" %}
  3. {% trans "Browser redirects back to the same page (next field)" %}
  4. {% trans "LocaleMiddleware reads the saved language on the next request" %}
  5. {% trans "trans tags resolve strings from the matching .mo file" %}
  6. {% trans "User sees the fully translated page" %}
